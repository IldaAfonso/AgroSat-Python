CREATE TABLE USUARIOS (
    ID_USUARIO NUMBER PRIMARY KEY,
    NOME VARCHAR2(100) NOT NULL,
    EMAIL VARCHAR2(100) NOT NULL UNIQUE,
    SENHA VARCHAR2(200) NOT NULL,
    PERFIL VARCHAR2(20) NOT NULL,
    
    CONSTRAINT CK_PERFIL
    CHECK (PERFIL IN ('ADMIN','GESTOR','OPERADOR'))
);

CREATE TABLE PROPRIEDADES (
    ID_PROPRIEDADE NUMBER PRIMARY KEY,
    NOME_PROPRIEDADE VARCHAR2(100) NOT NULL,
    LOCALIZACAO VARCHAR2(150) NOT NULL,
    TIPO_CULTIVO VARCHAR2(100),
    AREA_HECTARES NUMBER(10,2),
    
    ID_USUARIO NUMBER NOT NULL,

    CONSTRAINT FK_PROP_USUARIO
    FOREIGN KEY (ID_USUARIO)
    REFERENCES USUARIOS(ID_USUARIO)
);


CREATE TABLE SATELITES (
    ID_SATELITE NUMBER PRIMARY KEY,
    NOME_SATELITE VARCHAR2(100) NOT NULL,
    ORIGEM_DADOS VARCHAR2(100),
    TIPO_MONITORAMENTO VARCHAR2(100)
);


CREATE TABLE PROPRIEDADE_SATELITE (
    ID_PROPRIEDADE NUMBER,
    ID_SATELITE NUMBER,

    PRIMARY KEY (
        ID_PROPRIEDADE,
        ID_SATELITE
    ),

    CONSTRAINT FK_PS_PROP
    FOREIGN KEY (ID_PROPRIEDADE)
    REFERENCES PROPRIEDADES(ID_PROPRIEDADE),

    CONSTRAINT FK_PS_SAT
    FOREIGN KEY (ID_SATELITE)
    REFERENCES SATELITES(ID_SATELITE)
);


CREATE TABLE SENSORES (
    ID_SENSOR NUMBER PRIMARY KEY,
    TIPO_SENSOR VARCHAR2(50) NOT NULL,
    STATUS_SENSOR VARCHAR2(20),

    ID_PROPRIEDADE NUMBER NOT NULL,

    CONSTRAINT FK_SENSOR_PROP
    FOREIGN KEY (ID_PROPRIEDADE)
    REFERENCES PROPRIEDADES(ID_PROPRIEDADE),

    CONSTRAINT CK_STATUS_SENSOR
    CHECK (STATUS_SENSOR IN ('ATIVO','INATIVO'))
);

CREATE TABLE LEITURAS_CLIMATICAS (
    ID_LEITURA NUMBER PRIMARY KEY,
    TEMPERATURA NUMBER(5,2),
    UMIDADE NUMBER(5,2),
    INDICE_CHUVA NUMBER(5,2),
    RISCO_CLIMATICO VARCHAR2(20),
    DATA_LEITURA DATE,

    ID_SENSOR NUMBER NOT NULL,

    CONSTRAINT FK_LEITURA_SENSOR
    FOREIGN KEY (ID_SENSOR)
    REFERENCES SENSORES(ID_SENSOR),

    CONSTRAINT CK_RISCO
    CHECK (RISCO_CLIMATICO IN ('BAIXO','MEDIO','ALTO'))
);


CREATE TABLE ENERGIA (
    ID_ENERGIA NUMBER PRIMARY KEY,
    CONSUMO_KWH NUMBER(10,2),
    PRODUCAO_SOLAR NUMBER(10,2),
    EFICIENCIA_ENERGETICA NUMBER(5,2),
    DATA_REGISTRO DATE,

    ID_PROPRIEDADE NUMBER NOT NULL,

    CONSTRAINT FK_ENERGIA_PROP
    FOREIGN KEY (ID_PROPRIEDADE)
    REFERENCES PROPRIEDADES(ID_PROPRIEDADE)
);


CREATE TABLE CAMINHOES (
    ID_CAMINHAO NUMBER PRIMARY KEY,
    PLACA VARCHAR2(10) NOT NULL UNIQUE,
    MODELO VARCHAR2(100),
    CAPACIDADE_TON NUMBER(10,2),
    STATUS_OPERACAO VARCHAR2(20),

    CONSTRAINT CK_STATUS_CAMINHAO
    CHECK (
        STATUS_OPERACAO IN
        ('ATIVO','MANUTENCAO','INATIVO')
    )
);

CREATE TABLE ROTAS (
    ID_ROTA NUMBER PRIMARY KEY,
    ORIGEM VARCHAR2(100),
    DESTINO VARCHAR2(100),
    DISTANCIA_KM NUMBER(10,2),
    TEMPO_ESTIMADO NUMBER(10,2),
    RISCO_CLIMATICO VARCHAR2(20),

    ID_CAMINHAO NUMBER NOT NULL,

    CONSTRAINT FK_ROTA_CAMINHAO
    FOREIGN KEY (ID_CAMINHAO)
    REFERENCES CAMINHOES(ID_CAMINHAO)
);


CREATE TABLE ALERTAS (
    ID_ALERTA NUMBER PRIMARY KEY,
    TIPO_ALERTA VARCHAR2(50),
    DESCRICAO VARCHAR2(255),
    NIVEL_ALERTA VARCHAR2(20),
    DATA_ALERTA DATE,

    ID_PROPRIEDADE NUMBER NOT NULL,

    CONSTRAINT FK_ALERTA_PROP
    FOREIGN KEY (ID_PROPRIEDADE)
    REFERENCES PROPRIEDADES(ID_PROPRIEDADE),

    CONSTRAINT CK_NIVEL_ALERTA
    CHECK (
        NIVEL_ALERTA IN
        ('BAIXO','MEDIO','ALTO')
    )
);

SELECT *
FROM PROPRIEDADES;

SELECT
    NOME_PROPRIEDADE,
    AREA_HECTARES
FROM PROPRIEDADES
ORDER BY AREA_HECTARES DESC;

SELECT
    ID_SENSOR,
    TIPO_SENSOR,
    STATUS_SENSOR
FROM SENSORES
WHERE STATUS_SENSOR = 'ATIVO';


SELECT
    ID_ALERTA,
    TIPO_ALERTA,
    NIVEL_ALERTA,
    DATA_ALERTA
FROM ALERTAS
WHERE NIVEL_ALERTA = 'ALTO';


SELECT
    COUNT(*) AS TOTAL_PROPRIEDADES
FROM PROPRIEDADES;


SELECT
    SUM(CONSUMO_KWH) AS CONSUMO_TOTAL
FROM ENERGIA;


SELECT
    ROUND(AVG(EFICIENCIA_ENERGETICA),2) AS MEDIA_EFICIENCIA
FROM ENERGIA;


SELECT
    MAX(TEMPERATURA) AS MAIOR_TEMPERATURA
FROM LEITURAS_CLIMATICAS;


SELECT
    MIN(INDICE_CHUVA) AS MENOR_INDICE_CHUVA
FROM LEITURAS_CLIMATICAS;


SELECT
    ID_ENERGIA,
    CONSUMO_KWH,
    PRODUCAO_SOLAR,

    ROUND(
        (PRODUCAO_SOLAR / CONSUMO_KWH) * 100,
        2
    ) AS PERCENTUAL_EFICIENCIA

FROM ENERGIA;


SELECT
    P.ID_PROPRIEDADE,
    P.NOME_PROPRIEDADE,
    P.LOCALIZACAO,
    U.NOME AS RESPONSAVEL,
    U.EMAIL
FROM PROPRIEDADES P
INNER JOIN USUARIOS U
    ON P.ID_USUARIO = U.ID_USUARIO
ORDER BY P.NOME_PROPRIEDADE;


SELECT
    P.NOME_PROPRIEDADE,
    A.TIPO_ALERTA,
    A.NIVEL_ALERTA,
    A.DATA_ALERTA
FROM PROPRIEDADES P
INNER JOIN ALERTAS A
    ON P.ID_PROPRIEDADE = A.ID_PROPRIEDADE
ORDER BY A.DATA_ALERTA DESC;


SELECT
    C.PLACA,
    C.MODELO,
    R.ORIGEM,
    R.DESTINO,
    R.DISTANCIA_KM,
    R.RISCO_CLIMATICO
FROM CAMINHOES C
INNER JOIN ROTAS R
    ON C.ID_CAMINHAO = R.ID_CAMINHAO
ORDER BY R.DISTANCIA_KM DESC;


SELECT
    P.ID_PROPRIEDADE,
    P.NOME_PROPRIEDADE,
    P.LOCALIZACAO
FROM PROPRIEDADES P
LEFT JOIN ALERTAS A
    ON P.ID_PROPRIEDADE = A.ID_PROPRIEDADE
WHERE A.ID_ALERTA IS NULL;


SELECT
    S.ID_SATELITE,
    S.NOME_SATELITE,
    S.TIPO_MONITORAMENTO
FROM PROPRIEDADE_SATELITE PS
RIGHT JOIN SATELITES S
    ON PS.ID_SATELITE = S.ID_SATELITE
WHERE PS.ID_PROPRIEDADE IS NULL;
